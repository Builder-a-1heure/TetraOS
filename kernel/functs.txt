Structure du projet :
  boot/        Bootloader ASM, stage2, linker script
  drivers/     Pilotes matériels : VESA, VGA, souris PS/2, clavier, ATA
  gfx/         Rendu graphique : screen (scrollback + primitives), animation boot
  mem/         Gestion mémoire : allocateur boot, page frame allocator
  fs/          Système de fichiers RAY64
  ui/          Interface graphique : bureau, fenêtres, sessions/login
  shell/       main.c, shell.c (parse/dispatch), terminal.c (fenêtre + shell interactif),
               editor.c, tex.c
  lib/         Utilitaires C, I/O bas niveau, types globaux

================================================================================
  RÈGLES GÉNÉRALES FRAMEBUFFER
================================================================================

  ! IMPORTANT — Toute fonction qui écrit directement dans le framebuffer via
  ! vesa_put_pixel() ou gfx_fill_rect() bypasse le dirty tracker de vesa.c.
  ! Après un tel appel, il FAUT appeler vesa_invalidate_all() (ou
  ! vesa_invalidate_cell()) pour que vesa_draw_glyph() redessine correctement
  ! les glyphes concernés au prochain rendu.

  gfx_blend_pixel / gfx_*_blend : lisent le pixel existant (vesa_get_pixel)
  puis écrivent un mélange alpha ; même règle si la zone recouvrait du texte
  scrollback / glyphes — invalider si besoin après un bloc UI.

  Séquence type bureau / fenêtre (texte via glyphes + fond gfx) :
      screen_begin_ui();
          gfx_gradient_v / gfx_fill_rect ...
          vesa_invalidate_all();     // après gros gfx sur zone texte possible
          gfx_draw_text / vesa_draw_glyph ...
      screen_end_ui();

================================================================================
  drivers/vesa.c — Pilote framebuffer VESA
================================================================================

int  vesa_init(void)
  Lit la structure VESA laissée par stage2 à 0x9000 (magic, fb_addr, pitch,
  w, h, bpp). Initialise le framebuffer. Retourne 1 si OK, 0 sinon.
  Appel : une seule fois dans screen_init().

int  vesa_active(void)
  Retourne 1 si le mode VESA est actif, 0 si on est en VGA texte.
  Appel : vesa_active() — testé partout avant tout dessin pixel.

uint32_t vesa_width(void)
uint32_t vesa_height(void)
uint32_t vesa_pitch(void)
uint32_t vesa_fb_addr(void)
  Getters sur les dimensions et l'adresse du framebuffer.

int  vesa_text_cols(void)
int  vesa_text_rows(void)
  Nombre de colonnes/lignes de texte selon FONT_W/FONT_H et la résolution.

void vesa_put_pixel(int x, int y, uint32_t color)
  Écrit un pixel 32bpp directement dans le framebuffer. Bypass le dirty tracker.
  ! Appeler vesa_invalidate_cell(x/FONT_W, y/FONT_H) après si du texte était là.

uint32_t vesa_get_pixel(int x, int y)
  Lit un pixel depuis le framebuffer (utilisé par mouse_draw_cursor pour
  sauvegarder le fond avant de dessiner le curseur).

void vesa_fill(uint32_t color)
  Remplit tout le framebuffer d'une couleur + invalide tout le dirty buffer.
  Appel : vesa_fill(COLOR_BG) — utilisé dans clear_screen() et screen_exit_ui().

void vesa_draw_glyph(int px, int py, char c, uint32_t fg, uint32_t bg)
  Dessine un glyphe 8×16 en px/py pixels. Utilise le dirty tracker : si la
  cellule (px/FONT_W, py/FONT_H) a déjà ce char+fg+bg, ne redessine pas.
  Appel : vesa_draw_glyph(col*FONT_W, row*FONT_H, ch, fg, bg)

void vesa_invalidate_cell(int col, int row)
  Marque une cellule de la grille glyphe comme "à redessiner".
  Appel : vesa_invalidate_cell(x/FONT_W, y/FONT_H)

void vesa_invalidate_all(void)
  Invalide tout le dirty buffer — le prochain vesa_draw_glyph() redessinera
  toutes les cellules même si char+couleur semblent inchangés.
  ! À appeler après tout gfx_fill_rect() ou vesa_put_pixel() qui écrase du texte.

void vesa_clear_glyph(int col, int row, uint32_t bg)
  Efface une cellule en dessinant un espace avec la couleur de fond.

================================================================================
  drivers/mouse.c — Pilote souris PS/2
================================================================================

void mouse_init(void)
  Initialise le contrôleur 8042 : active le port auxiliaire, désactive IRQ12
  (polling pur), reset la souris, active le streaming.
  Appel : mouse_init() dans kmain(), après screen_init().

int  mouse_poll(void)
  Polling non-bloquant : lit un octet si disponible (bit0 du status 0x64).
  Accumule les 3 octets d'un paquet PS/2. Retourne 1 quand un paquet complet
  est décodé et g_mouse mis à jour, 0 sinon.
  Resynchronisation via bit3 du premier octet (toujours 1 dans un paquet souris).
  Le commentaire dans mouse.c précise que le bit 5 du status (souris) n'est
  pas fiable sur toutes les VM — le routage correct des octets 2 et 3 est
  assuré côté input_dispatch_char() via mouse_in_packet().
  Appel : while (mouse_poll()) mouse_moved = 1;

int  mouse_in_packet(void)
  Retourne 1 si on est en cours de réception d'un paquet (octets 2 ou 3).
  Utilisé en tête de input_dispatch_char() : tant qu'un paquet est incomplet,
  tout octet en file doit aller à mouse_poll(), même si le bit 5 du port 0x64
  est à 0 (QEMU). Évite de traiter un octet souris comme scancode clavier.

void mouse_draw_cursor(void)
  Sauvegarde les pixels sous le curseur dans g_cursor_bg[], puis dessine
  la flèche (12×19 pixels, bitmap 1bpp). À appeler après mouse_erase_cursor().
  Appel : mouse_draw_cursor() après tout redraw du fond.

void mouse_erase_cursor(void)
  Restaure les pixels sauvegardés (efface la flèche), puis invalide les cellules
  de la grille glyphe recouvertes pour forcer leur redraw par vesa_draw_glyph().
  ! Toujours appeler mouse_erase_cursor() AVANT de redessiner le fond.
  Appel : mouse_erase_cursor() avant tout gfx_fill_rect() / desktop_redraw_all().

void mouse_reset_cursor(void)
  Invalide le curseur sauvegardé sans restaurer les pixels. À appeler après
  vesa_fill() ou screen_exit_ui() pour éviter de restaurer un fond périmé.
  Appel : automatiquement dans screen_exit_ui().

  Accès à l'état courant : extern MouseState g_mouse;
    g_mouse.x, g_mouse.y       — position en pixels
    g_mouse.btn_left/right/middle — état des boutons (1=enfoncé)

================================================================================
  drivers/input.c — Clavier PS/2 + dispatcher 8042
================================================================================

void input_set_mouse_packet_handler(void (*fn)(void))
  Enregistre un callback appelé lorsqu'un paquet souris complet est reçu
  (après mouse_poll() qui retourne 1). Si fn est NULL, comportement par défaut :
  mouse_erase_cursor() + mouse_draw_cursor().
  Utilisé par terminal_set_mouse_handler() (shell/terminal.c) pour brancher
  on_term_mouse() du bureau pendant terminal_run() — drag de fenêtre, hover
  titlebar, sans dupliquer la lecture du port 0x60.

char input_dispatch_char(void)
  DISPATCHER CENTRALISÉ. Boucle interne sur le status 0x64 (bit0 = donnée prête) :
    1) Si mouse_in_packet() : mouse_poll() ; si paquet complet → handler
       souris (voir input_set_mouse_packet_handler) ou redraw curseur par défaut ;
       retourne 0.
    2) Sinon si bit5=1 (donnée souris) : idem mouse_poll() + handler/curseur ;
       retourne 0.
    3) Sinon : lit 0x60 comme scancode clavier, gère Shift/Ctrl, flèches,
       retourne le caractère ou 0 (modificateur / relâchement).
  C'est la brique de base pour clavier + souris sans double consommation
  d'octets sur le port 0x60.
  Appel : char c = input_dispatch_char(); if (c) { ... }

char keyboard_get_char(void)
  Bloque jusqu'à obtenir un vrai caractère clavier (ignore les événements souris
  mais les traite au passage via input_dispatch_char).
  Appel : char c = keyboard_get_char();

char get_input_char(void)
  Version simplifiée sans gestion souris. Utilisée dans les contextes purement
  texte (avant l'init de la souris).

int  keyboard_read_scancode(void)
  Lit un scancode brut depuis le port 0x60 (bloquant). Usage interne.

void handle_input(void)
  Boucle de saisie bloquante pour le mode shell texte (non-VESA). Usage interne.

  Variables globales accessibles :
    extern int shift_pressed;   — état de la touche Shift
    extern int ctrl_pressed;    — état de la touche Ctrl
    extern char input_buffer[]; — buffer de saisie courant
    extern int  input_index;    — indice d'écriture dans input_buffer

================================================================================
  gfx/screen.c — Scrollback texte + primitives graphiques
================================================================================

--- Initialisation ---

void screen_init(void)
  Init VESA (ou VGA en fallback), fixe g_text_cols/g_text_rows.
  Appel : première chose dans kmain().

--- Cycle de rendu UI ---

void screen_begin_ui(void)
  g_ui_drawing = 1 : screen_render() ne flushe plus tout de suite (marque seulement
  g_screen_dirty si appelé). Les gfx_* écrivent directement dans le framebuffer.

void screen_end_ui(void)
  g_ui_drawing = 0 ; si g_screen_dirty alors render_vesa()/render_vga() (réaffiche
  le scrollback par-dessus). Le bureau dessine en gfx sans passer par screen_render,
  donc souvent pas de flush ici — le contenu graphique reste tel quel sur le FB.

void screen_exit_ui(void)
  Quitte l'UI graphique pour revenir au terminal texte : vesa_fill(BG),
  invalide tout, appelle mouse_reset_cursor(). Utilisé à la fin du login.

void screen_flush(void)
  Force un rendu immédiat sans passer par le dirty flag.

void screen_invalidate(void)
  Marque l'écran comme devant être redessiné (sans redessiner immédiatement).

--- Scrollback texte ---

void print_char(char c)
  Écrit un caractère dans le scrollback (ou redirige vers g_print_hook si
  le terminal fenêtré est actif).

void print_string(const char* str)

void print_int(int num)
void print_hex(uint32_t num)
void print_dec(uint32_t num)
  Formatage numérique vers le scrollback.

void clear_screen(void)

void screen_clear_visible(void)
  Vide uniquement la zone visible (sans toucher à l'historique scrollback).

void screen_scroll_up(void)
void screen_scroll_down(void)
void screen_goto_bottom(void)
int  screen_at_bottom(void)
  Navigation dans le scrollback.

void set_cursor(int row, int col)

    Si non-NULL, print_char()/print_string() appellent g_print_hook(c) au lieu

  ui/desktop.c — Bureau graphique et terminal fenêtré

void desktop_run(void)
  Boucle principale du bureau (tant que logged_in). Gère :
    - Fond dégradé + barre des tâches + icône Terminal
    - Polling souris (mouse_poll) + curseur hors fenêtre terminal bloquante
    - Clic sur l'icône → open_terminal() qui appelle terminal_run() et bloque
      jusqu'à fermeture [X] ou logout
  Important : pendant terminal_run(), cette boucle ne s'exécute pas ; le drag
  de fenêtre et le hover titlebar passent par input_dispatch_char → callback
  souris (voir terminal_set_mouse_handler / on_term_mouse).
  Retourne quand g_session_manager.logged_in passe à 0 (logout).
  Appel : desktop_run() dans kmain() après session_do_login_flow().

  Fonctions internes (static) :

  draw_background()         Dégradé plein écran + vesa_invalidate_all() pour
                            resynchroniser le dirty tracker avec le framebuffer
                            (évite texte fantôme après gfx_* sur le bureau).
  draw_taskbar()            Barre du bas (session, titre)
  draw_icon()               Icône Terminal ; libellé sous l'icône avec fond
                            échantillonné via vesa_get_pixel (cohérent avec le fond)
  draw_window(focused)      Chrome fenêtre : ombre décalée, fond WIN_BG, bordures
                            « vitrées » (gfx_stroke_rect_blend + highlight),
                            paint_titlebar_chrome(), séparateur titlebar/contenu
  paint_titlebar_chrome(tb) Remplit la barre de titre + texte « Terminal » + bouton [X]
  draw_terminal_content(full_sync)
                            Rendu du buffer g_term dans la zone texte. full_sync=1
                            après redraw complet de fenêtre (vesa_invalidate_all) ;
                            full_sync=0 : incrémental (cache VESA) + invalidation
                            cellule curseur (gfx_fill_rect curseur hors dirty glyph).
  desktop_redraw_all()      Fond + taskbar + icône OU fenêtre + contenu terminal
  term_redraw_content()     Redraw partiel : draw_terminal_content(0) seulement
  open_terminal()           g_win_open, dimensions, terminal_init, callbacks redraw/
                            souris, desktop_redraw_all, terminal_run() bloquant
  close_terminal()          Ferme la fenêtre, retire hooks
  on_term_redraw()          Callback → term_redraw_content()
  on_term_mouse()           Début drag (s_term_prev_left + hit_window_drag_region),
                            déplacement fenêtre, clamp écran, ou repaint titlebar
  hit_titlebar / hit_close / hit_window_drag_region
                            Hit-tests : drag depuis titlebar (hors zone [X]) ou
                            bande WIN_DRAG_BORDER sur les côtés et le bas (hors
                            zone texte centrale)

  API shell/terminal.h (buffer g_term[], utilisé par desktop pour le rendu) :

  void terminal_init(uint32_t fg, uint32_t bg)
  void terminal_putchar(char c)   void terminal_puts(const char* s)
  char terminal_getchar(void)     Boucle sur input_dispatch_char() ; scroll flèches
  void terminal_run(void)         Installe g_print_hook = terminal_putchar, bannière,
                                  boucle prompt → shell_parse / shell_dispatch / cmd_sl
  void terminal_set_redraw_fn(void (*fn)(void))
  void terminal_set_mouse_handler(TermMouseHandlerFn fn)  → input_set_mouse_packet_handler
  void terminal_scroll_up(void)   void terminal_scroll_down(void)

  Variables globales (terminal.c) : g_term[][], g_term_row, g_term_col, g_term_view,
    g_term_cols_vis, g_term_rows_vis, g_term_fg, g_term_bg

int  shell_parse(const char* input,
                 char* mod_out, int mod_max,
                 char* cmd_out, int cmd_max,

  Parse une commande "module.commande args" en trois parties.

void shell_dispatch(const char* mod, const char* cmd, const char* args)

void dispatch_session(const char* cmd, const char* args)
  Traite les commandes du namespace "session" (adduser, deluser, whoami, etc.)

void dispatch_sys(const char* cmd, const char* args)
  Traite les commandes du namespace "sys" (clear, shutdown, help, etc.)
